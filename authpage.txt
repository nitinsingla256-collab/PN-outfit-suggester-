    23	type AuthMode = "signin" | "signup" | "forgot";
    24	
    25	function sanitizeAuthError(err: any): string {
    26	  const msg = typeof err === 'string' ? err : err?.message || '';
    27	  if (!msg) return 'Authentication encountered a temporary issue. Please try again.';
    28	  if (
    29	    msg.includes('<!doctype') ||
    30	    msg.includes('Unexpected token') ||
    31	    msg.includes('not valid JSON') ||
    32	    msg.includes('502') ||
    33	    msg.includes('503')
    34	  ) {
    35	    return 'Authentication service momentarily reconnecting. You can sign in smoothly.';
    36	  }
    37	  return msg;
    38	}

   338	                        />
